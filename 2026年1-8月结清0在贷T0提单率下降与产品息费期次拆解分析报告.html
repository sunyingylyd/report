/* 执行环境：PostgreSQL；本模块会创建后续模块所需的临时表。 */
SET TIME ZONE 'America/Belize';

/* 1. cohort：结清日期在 2026-01-01 至 2026-10-28 的正常结清放款订单。 */
CREATE TEMP TABLE settled_orders AS
SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.repaid_date::date AS repaid_date
FROM wangchuanliang.order_loan_f_v2_copy AS o
WHERE o.is_remit = 1
  AND o.user_id <> 5101518
  AND o.loan_status_code = 7
  AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND o.repaid_date >= DATE '2026-01-01'
  AND o.repaid_date <= DATE '2026-10-28';
CREATE INDEX ON settled_orders (user_id, repaid_date, repaid_time);

/* 2. 仅拉取 cohort 用户的订单，并以结清时点无其他在贷订单判定“结清 0 在贷”。 */
CREATE TEMP TABLE candidate_users AS SELECT DISTINCT user_id FROM settled_orders;
CREATE INDEX ON candidate_users (user_id);
CREATE TEMP TABLE all_orders AS
SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.due_date,
       o.remit_amt, o.repaid_amt, o.pre_amt, o.post_amt, o.total_term,
       o.product_no, o.loan_day, o.loan_status_code, o.is_remit
FROM wangchuanliang.order_loan_f_v2_copy AS o
JOIN candidate_users AS u ON u.user_id = o.user_id;
CREATE INDEX ON all_orders (user_id, apply_time);
CREATE INDEX ON all_orders (user_id, repaid_time);

CREATE TEMP TABLE clean_settles AS
SELECT s.*
FROM settled_orders AS s
WHERE NOT EXISTS (
  SELECT 1
  FROM all_orders AS o
  WHERE o.user_id = s.user_id
    AND o.serial_id <> s.serial_id
    AND o.apply_time < s.repaid_time
    AND (o.loan_status_code IN (5, 8)
         OR (o.loan_status_code = 7 AND o.repaid_time > s.repaid_time))
);
CREATE INDEX ON clean_settles (user_id, repaid_date, repaid_time);

/* 3. 在结清当日或前一日读取获额通过记录；同一用户 20 秒内多次获额只保留 serial_id 最大的一笔。 */
CREATE TEMP TABLE credit_days AS
SELECT DISTINCT user_id, repaid_date - 1 AS vir_date FROM clean_settles
UNION
SELECT DISTINCT user_id, repaid_date AS vir_date FROM clean_settles;
CREATE INDEX ON credit_days (user_id, vir_date);

CREATE TEMP TABLE raw_credit_pass AS
SELECT DISTINCT ON (v.user_id, v.serial_id)
       v.user_id, v.serial_id, v.vir_date::date AS vir_date,
       (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize') AS vir_time
FROM wangchuanliang.order_vir_f_copy AS v
JOIN credit_days AS d ON d.user_id = v.user_id AND d.vir_date = v.vir_date
WHERE v.is_pass1 = 1
ORDER BY v.user_id, v.serial_id, v.vir_unix DESC;
CREATE INDEX ON raw_credit_pass (user_id, vir_time);

CREATE TEMP TABLE dedup_credit_pass AS
WITH ordered AS (
  SELECT *, LAG(vir_time) OVER (PARTITION BY user_id ORDER BY vir_time, serial_id) AS prev_vir_time
  FROM raw_credit_pass
), batched AS (
  SELECT *, SUM(CASE WHEN prev_vir_time IS NULL
                            OR vir_time - prev_vir_time > INTERVAL '20 seconds'
                       THEN 1 ELSE 0 END)
            OVER (PARTITION BY user_id ORDER BY vir_time, serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
  FROM ordered
)
SELECT user_id, serial_id, vir_date, vir_time
FROM (
  SELECT *, ROW_NUMBER() OVER (PARTITION BY user_id, batch_id ORDER BY serial_id DESC) AS rn
  FROM batched
) AS x
WHERE rn = 1;
CREATE INDEX ON dedup_credit_pass (user_id, vir_date, vir_time);

/* 4. 同一用户同一结清日仅保留结清后最早的有效获额。 */
CREATE TEMP TABLE t0_samples AS
SELECT user_id, repaid_date, credit_vir_time
FROM (
  SELECT c.user_id, c.repaid_date, cp.vir_time AS credit_vir_time,
         ROW_NUMBER() OVER (
           PARTITION BY c.user_id, c.repaid_date
           ORDER BY cp.vir_time ASC, cp.serial_id DESC, c.repaid_time ASC
         ) AS rn
  FROM clean_settles AS c
  JOIN dedup_credit_pass AS cp
    ON cp.user_id = c.user_id
   AND cp.vir_date = c.repaid_date
   AND cp.vir_time >= c.repaid_time
) AS x
WHERE rn = 1;
CREATE INDEX ON t0_samples (user_id, repaid_date, credit_vir_time);

/* 5. 每个 T0 样本只取获额后 30 天内最早的一次提单。 */
CREATE TEMP TABLE first_apply AS
SELECT user_id, repaid_date, first_apply_serial_id, first_apply_date,
       first_apply_is_remit, first_apply_remit_amt, first_apply_pre_amt,
       first_apply_post_amt, first_apply_total_term
FROM (
  SELECT t.user_id, t.repaid_date,
         a.serial_id AS first_apply_serial_id,
         a.apply_time::date AS first_apply_date,
         a.is_remit AS first_apply_is_remit,
         a.remit_amt AS first_apply_remit_amt,
         a.pre_amt AS first_apply_pre_amt,
         a.post_amt AS first_apply_post_amt,
         a.total_term AS first_apply_total_term,
         ROW_NUMBER() OVER (
           PARTITION BY t.user_id, t.repaid_date
           ORDER BY a.apply_time ASC, a.serial_id ASC
         ) AS rn
  FROM t0_samples AS t
  JOIN all_orders AS a
    ON a.user_id = t.user_id
   AND a.apply_time >= t.credit_vir_time
   AND a.apply_time < t.credit_vir_time + INTERVAL '31 days'
   AND a.apply_time < TIMESTAMP '2026-10-29 00:00:00'
) AS x
WHERE rn = 1;
CREATE INDEX ON first_apply (user_id, repaid_date);

/* 模块2A：各累计指标分别限定T0事件截止日，分母不能互换。
   统一观察首次提单至2026-10-28结束。7/15/30天累计包含当天。
   D0截至10/28；D7截至10/20；D15截至10/12；D30截至09/28。
   10月D30无成熟分母，返回NULL，不能补0或外推。 */
WITH events AS (
 SELECT t.user_id,t.repaid_date,
        f.first_apply_date-t.repaid_date AS first_apply_day
 FROM t0_samples t LEFT JOIN first_apply f USING(user_id,repaid_date)
), monthly AS (
 SELECT TO_CHAR(DATE_TRUNC('month',repaid_date),'YYYY-MM') AS month,
   COUNT(*) AS sample_cnt,
   MAX(repaid_date) AS day0_event_end,
   SUM(CASE WHEN first_apply_day=0 THEN 1 ELSE 0 END) AS day0_cnt,
   SUM(CASE WHEN repaid_date<=DATE '2026-10-20' THEN 1 ELSE 0 END) AS day7_sample_cnt,
   SUM(CASE WHEN repaid_date<=DATE '2026-10-20' AND first_apply_day BETWEEN 0 AND 7 THEN 1 ELSE 0 END) AS day7_cnt,
   MAX(CASE WHEN repaid_date<=DATE '2026-10-20' THEN repaid_date END) AS day7_event_end,
   SUM(CASE WHEN repaid_date<=DATE '2026-10-20' AND first_apply_day BETWEEN 1 AND 7 THEN 1 ELSE 0 END) AS day1_7_cnt,
   SUM(CASE WHEN repaid_date<=DATE '2026-10-12' THEN 1 ELSE 0 END) AS day15_sample_cnt,
   SUM(CASE WHEN repaid_date<=DATE '2026-10-12' AND first_apply_day BETWEEN 0 AND 15 THEN 1 ELSE 0 END) AS day15_cnt,
   MAX(CASE WHEN repaid_date<=DATE '2026-10-12' THEN repaid_date END) AS day15_event_end,
   SUM(CASE WHEN repaid_date<=DATE '2026-10-12' AND first_apply_day BETWEEN 8 AND 15 THEN 1 ELSE 0 END) AS day8_15_cnt,
   SUM(CASE WHEN repaid_date<=DATE '2026-09-28' THEN 1 ELSE 0 END) AS day30_sample_cnt,
   SUM(CASE WHEN repaid_date<=DATE '2026-09-28' AND first_apply_day BETWEEN 0 AND 30 THEN 1 ELSE 0 END) AS day30_cnt,
   MAX(CASE WHEN repaid_date<=DATE '2026-09-28' THEN repaid_date END) AS day30_event_end,
   SUM(CASE WHEN repaid_date<=DATE '2026-09-28' AND first_apply_day BETWEEN 16 AND 30 THEN 1 ELSE 0 END) AS day16_30_cnt
 FROM events GROUP BY 1
)
SELECT *,
 ROUND(day0_cnt::numeric/NULLIF(sample_cnt,0)*100,2) AS day0_rate_pct,
 ROUND(day7_cnt::numeric/NULLIF(day7_sample_cnt,0)*100,2) AS day7_rate_pct,
 ROUND(day15_cnt::numeric/NULLIF(day15_sample_cnt,0)*100,2) AS day15_rate_pct,
 ROUND(day30_cnt::numeric/NULLIF(day30_sample_cnt,0)*100,2) AS day30_rate_pct,
 ROUND(day1_7_cnt::numeric/NULLIF(day7_sample_cnt,0)*100,2) AS day1_7_rate_pct,
 ROUND(day8_15_cnt::numeric/NULLIF(day15_sample_cnt,0)*100,2) AS day8_15_rate_pct,
 ROUND(day16_30_cnt::numeric/NULLIF(day30_sample_cnt,0)*100,2) AS day16_30_rate_pct
FROM monthly ORDER BY month;

/* 模块2B：共同成熟样本的阶段拆解。全部阶段使用相同分母，方可相加。
   仅纳入截至2026-09-28的T0事件；9月是1—28日，10月不展示。 */
WITH monthly AS (
 SELECT TO_CHAR(DATE_TRUNC('month',t.repaid_date),'YYYY-MM') AS month,
   MAX(t.repaid_date) AS event_end,COUNT(*) AS sample_cnt,
   SUM(CASE WHEN f.first_apply_date-t.repaid_date=0 THEN 1 ELSE 0 END) AS day0_cnt,
   SUM(CASE WHEN f.first_apply_date-t.repaid_date BETWEEN 1 AND 7 THEN 1 ELSE 0 END) AS day1_7_cnt,
   SUM(CASE WHEN f.first_apply_date-t.repaid_date BETWEEN 8 AND 15 THEN 1 ELSE 0 END) AS day8_15_cnt,
   SUM(CASE WHEN f.first_apply_date-t.repaid_date BETWEEN 16 AND 30 THEN 1 ELSE 0 END) AS day16_30_cnt
 FROM t0_samples t LEFT JOIN first_apply f USING(user_id,repaid_date)
 WHERE t.repaid_date<=DATE '2026-09-28' GROUP BY 1
)
SELECT *,
 ROUND(day0_cnt::numeric/NULLIF(sample_cnt,0)*100,2) AS day0_rate_pct,
 ROUND(day1_7_cnt::numeric/NULLIF(sample_cnt,0)*100,2) AS day1_7_rate_pct,
 ROUND(day8_15_cnt::numeric/NULLIF(sample_cnt,0)*100,2) AS day8_15_rate_pct,
 ROUND(day16_30_cnt::numeric/NULLIF(sample_cnt,0)*100,2) AS day16_30_rate_pct,
 ROUND((day0_cnt+day1_7_cnt+day8_15_cnt+day16_30_cnt)::numeric/NULLIF(sample_cnt,0)*100,2) AS day30_rate_pct
FROM monthly ORDER BY month;


/* 先执行模块 1。统计当天最早提单订单，不以是否成功放款作为筛选条件。
   加权息费率 = SUM(pre_amt + post_amt) / SUM(remit_amt)。 */
WITH day0_remitted AS (
  SELECT f.*
  FROM t0_samples AS t
  JOIN first_apply AS f USING (user_id, repaid_date)
  WHERE f.first_apply_date = t.repaid_date
    AND f.first_apply_remit_amt > 0
), term_monthly AS (
  SELECT DATE_TRUNC('month', repaid_date) AS month_date,
         first_apply_total_term AS total_term,
         COUNT(DISTINCT first_apply_serial_id) AS order_cnt,
         SUM(COALESCE(first_apply_pre_amt,0)+COALESCE(first_apply_post_amt,0)) AS fee_amt,
         SUM(first_apply_remit_amt) AS principal_amt,
         ROUND(SUM(COALESCE(first_apply_pre_amt, 0) + COALESCE(first_apply_post_amt, 0))::numeric
               / NULLIF(SUM(first_apply_remit_amt), 0) * 100, 2) AS term_fee_rate_pct
  FROM day0_remitted
  GROUP BY 1, 2
), term_with_share AS (
  SELECT TO_CHAR(month_date, 'YYYY-MM') AS month, total_term, order_cnt, fee_amt,principal_amt,term_fee_rate_pct,
         ROUND(principal_amt::numeric/NULLIF(SUM(principal_amt) OVER(PARTITION BY month_date),0)*100,2) AS principal_share_pct,
         ROUND(order_cnt::numeric / SUM(order_cnt) OVER (PARTITION BY month_date) * 100, 2) AS term_share_pct
  FROM term_monthly
), overall AS (
  SELECT TO_CHAR(DATE_TRUNC('month', repaid_date), 'YYYY-MM') AS month,
         ROUND(SUM(COALESCE(first_apply_pre_amt, 0) + COALESCE(first_apply_post_amt, 0))::numeric
               / NULLIF(SUM(first_apply_remit_amt), 0) * 100, 2) AS overall_fee_rate_pct
  FROM day0_remitted
  GROUP BY 1
)
SELECT t.month,t.total_term,t.order_cnt,t.fee_amt,t.principal_amt,t.principal_share_pct,t.term_share_pct,
       t.term_fee_rate_pct, o.overall_fee_rate_pct
FROM term_with_share AS t
JOIN overall AS o USING (month)
ORDER BY t.month, t.total_term;

def padded_range(values, minimum=None, maximum=None, tick=5):
    values = [float(v) for v in values if v is not None]

def segment_hits_rect(x1, y1, x2, y2, rect):
    """Liang–Barsky segment/rectangle intersection, including boundaries."""
    left, top, right, bottom = rect
    dx, dy = x2-x1, y2-y1
    p, q = [-dx, dx, -dy, dy], [x1-left, right-x1, y1-top, bottom-y1]
    lo, hi = 0.0, 1.0
    for pi, qi in zip(p, q):
        if pi == 0:
            if qi < 0:
                return False
        elif pi < 0:
            lo = max(lo, qi/pi)
        else:
            hi = min(hi, qi/pi)
    return lo <= hi

    all_values = [v for s in series if s.get('axis') != 'right' for v in s['values'] if v is not None]
    lo, hi = y_range or padded_range(all_values)
    ranges = {'left': (lo, hi), 'right': right_range or (lo, hi)}
    # Keep endpoint labels clear of both percentage axes.
    xs = [left + 40 + (pw - 80) * i / max(1, len(labels) - 1) for i in range(len(labels))]

    def y(value, axis='left'):
        low, high = ranges[axis]
        return top + (high - value) / (high - low) * ph

    svg = [f'<svg id="{e(chart_id)}" class="chart line-chart" data-axis-min="{lo}" data-axis-max="{hi}" viewBox="0 0 {width} {height}" role="img" '
           f'aria-labelledby="{e(chart_id)}-title {e(chart_id)}-desc">',
           f'<title id="{e(chart_id)}-title">{e(title)}</title>',
           f'<desc id="{e(chart_id)}-desc">全部有效数据点均显示数值。缺失成熟样本的月份不绘制。</desc>',
           f'<rect width="{width}" height="{height}" rx="12" fill="{BG}"/>',
           f'<text class="axis-title" x="{left}" y="30">{e(left_title)}</text>']
    if right_title:
        svg.append(f'<text class="axis-title" text-anchor="end" x="{width-right}" y="30">{e(right_title)}</text>')
    for i in range(6):
        value = lo + (hi - lo) * i / 5
        yy = y(value)
        svg += [f'<line class="gridline" x1="{left}" y1="{yy:.1f}" x2="{width-right}" y2="{yy:.1f}"/>',
                f'<text class="axis-label" x="{left-15}" y="{yy+5:.1f}" text-anchor="end">{value:g}%</text>']
        if right_range:
            rv = right_range[0] + (right_range[1] - right_range[0]) * i / 5
            svg.append(f'<text class="axis-label" x="{width-right+15}" y="{yy+5:.1f}">{rv:g}%</text>')
    for x, label in zip(xs, labels):
        parts = label.split('|')
        svg.append(f'<text class="month-label" x="{x:.1f}" y="{height-bottom+37}" text-anchor="middle">{e(parts[0])}</text>')
        if len(parts) > 1:
            svg.append(f'<text class="partial-label" x="{x:.1f}" y="{height-bottom+59}" text-anchor="middle">{e(parts[1])}</text>')
    segments, label_boxes = [], []
    for s in series:
        chunks, current = [], []
        for x, v in zip(xs, s['values']):
            if v is None:
                if current:
                    chunks.append(current)
                    current = []
            else:
                current.append((x, y(v, s.get('axis', 'left'))))
        if current:
            chunks.append(current)
        for chunk in chunks:
            coordinates = ' '.join(f'{x:.1f},{yy:.1f}' for x, yy in chunk)
            svg.append(f'<polyline class="data-series" data-series="{e(s["name"])}" points="{coordinates}" fill="none" stroke="{s["color"]}" stroke-width="3.4" stroke-linejoin="round"/>')
            segments.extend((a[0],a[1],b[0],b[1]) for a,b in zip(chunk,chunk[1:]))
        for index, (x, v) in enumerate(zip(xs, s['values'])):
            if v is not None:
                yy = y(v, s.get('axis', 'left'))
                svg.append(f'<circle class="data-marker" data-series="{e(s["name"])}" data-month="2026-{index+1:02}" cx="{x:.1f}" cy="{yy:.1f}" r="4.3" fill="{s["color"]}" stroke="{BG}" stroke-width="1.5"><title>{e(s["name"])}：{pct(v)}</title></circle>')
    # Search above and below each point at 2 px intervals. Validate against all
    # series segments and all markers, including neighboring months.
    marker_points = [(x,y(v,s.get('axis','left'))) for s in series
                     for x,v in zip(xs,s['values']) if v is not None]
    for s in series:
        for index, x in enumerate(xs):
            value = s['values'][index]
            if value is None:
                continue
            actual = y(value,s.get('axis','left'))
            direction = 1 if s.get('label_offset',-20)>0 else -1
            candidates = [actual+sign*offset for offset in range(20,501,2)
                          for sign in [direction,-direction]]
            best = None
            for cy in candidates:
                if not top-22 <= cy <= height-bottom+16:
                    continue
                box = (x-35,cy-11,x+35,cy+11)
                expanded=(box[0]-4,box[1]-4,box[2]+4,box[3]+4)
                if any(segment_hits_rect(*seg,expanded) for seg in segments):
                    continue
                if any(box[0]-8<=px<=box[2]+8 and box[1]-8<=py<=box[3]+8 for px,py in marker_points):
                    continue
                if any(rects_overlap(box,other) for other in label_boxes):
                    continue
                best=(cy,box)
                break
            if best is None:
                raise ValueError(f'Cannot place collision-free label: {chart_id}, {index+1}, {s["name"]}')
            cy,box=best
            label_boxes.append(box)
            attrs=f'data-series="{e(s["name"])}" data-month="2026-{index+1:02}"'
            svg.append(f'<rect class="data-label-bg" {attrs} x="{box[0]:.1f}" y="{box[1]:.1f}" width="70" height="22" rx="5" fill="{BG}"/>')
            svg.append(f'<text class="point-label data-label" {attrs} x="{x:.1f}" y="{cy+5:.1f}" fill="{s["color"]}" text-anchor="middle">{pct(value)}</text>')
    line_collisions=sum(segment_hits_rect(*seg,(b[0]-3,b[1]-3,b[2]+3,b[3]+3)) for b in label_boxes for seg in segments)
    marker_collisions=sum(b[0]-7<=px<=b[2]+7 and b[1]-7<=py<=b[3]+7 for b in label_boxes for px,py in marker_points)
    label_collisions=sum(rects_overlap(a,b,3) for i,a in enumerate(label_boxes) for b in label_boxes[i+1:])
    assert line_collisions == marker_collisions == label_collisions == 0
    CHART_QA.append(dict(chart_id=chart_id,label_count=len(label_boxes),marker_count=len(marker_points),
                         line_collisions=line_collisions,marker_collisions=marker_collisions,
                         label_collisions=label_collisions,leader_lines=0))
    svg.append('</svg>')
    return '<div class="chart-scroll">' + ''.join(svg) + '</div>'


def stack_chart(monthly, terms, term_keys, term_colors):
    width, height = 1320, 620
    left, right, top, bottom = 90, 45, 90, 118
    pw, ph = width-left-right, height-top-bottom
    band, bar = pw / len(monthly), 69
    rows = {(t['month'], t['total_term']): t for t in terms}
    svg = [f'<svg class="chart" viewBox="0 0 {width} {height}" role="img" aria-label="当天首次提单且本金大于零订单的期次结构与各期次加权息费">',
           f'<rect width="{width}" height="{height}" rx="12" fill="{BG}"/>',
           '<text class="axis-title" x="90" y="30">订单占比（柱高）；柱内数字为该期次加权息费</text>']
    for value in [0, 25, 50, 75, 100]:
        yy = top + (100-value)/100*ph
        svg += [f'<line class="gridline" x1="{left}" y1="{yy}" x2="{width-right}" y2="{yy}"/>',
                f'<text class="axis-label" x="{left-12}" y="{yy+5}" text-anchor="end">{value}%</text>']
    for i, m in enumerate(monthly):
        x = left + band*(i+.5)
        cumulative = 0
        svg.append(f'<text class="bar-count" x="{x:.1f}" y="{top-23}" text-anchor="middle">{n(m["fee_order_cnt"])}单</text>')
        for term in term_keys:
            t = rows.get((m['month'], term))
            if not t:
                continue
            share = float(t['order_cnt'])/float(m['fee_order_cnt'])*100 if m['fee_order_cnt'] else 0
            h = share/100*ph
            yy = top+ph-(cumulative+share)/100*ph
            color = term_colors[term]
            svg.append(f'<rect x="{x-bar/2:.1f}" y="{yy:.2f}" width="{bar}" height="{h:.2f}" fill="{color}" stroke="{BG}" stroke-width="1"><title>{e(term_name(term))}：订单占比 {pct(t["term_share_pct"])}，加权息费 {pct(t["term_fee_rate_pct"])}</title></rect>')
            if h >= 30:
                svg.append(f'<text class="stack-value" x="{x:.1f}" y="{yy+h/2+5:.1f}" text-anchor="middle">{pct(t["term_fee_rate_pct"])}</text>')
            cumulative += share
        parts = month_label(m).split('|')
        svg.append(f'<text class="month-label" x="{x:.1f}" y="{height-bottom+38}" text-anchor="middle">{parts[0]}</text>')
        if len(parts) > 1:
            svg.append(f'<text class="partial-label" x="{x:.1f}" y="{height-bottom+60}" text-anchor="middle">{parts[1]}</text>')
    svg.append('</svg>')
    return '<div class="chart-scroll">' + ''.join(svg) + '</div>'
